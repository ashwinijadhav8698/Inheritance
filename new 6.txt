Q12. Create a class Employee with:
Fields: id, name, basicSalary, isPermanent
Parameterized constructor to initialize fields.
Write methods:
calculateNetSalary() →
if permanent → HRA = 20%, DA = 10%, PF = 12%
if temporary → HRA = 10%, DA = 5%, PF = 0
compareSalary(Employee e) → return higher salary employee name.
Task Logic:
 Create 3 Employee objects.
 Compare each pair using logic (nested comparison).
Expected Learning: Complex if-else + constructor initialization + inter-object comparison.

class Employee {
    int id;
    String name;
    double basicSalary;
    boolean isPermanent;

    Employee(int id, String name, double basicSalary, boolean isPermanent) {
        this.id = id;
        this.name = name;
        this.basicSalary = basicSalary;
        this.isPermanent = isPermanent;
    }

    double calculateNetSalary() {
        double hra, da, pf;

        if (isPermanent) {
            hra = 0.20 * basicSalary;
            da = 0.10 * basicSalary;
            pf = 0.12 * basicSalary;
        } else {
            hra = 0.10 * basicSalary;
            da = 0.05 * basicSalary;
            pf = 0;
        }

        return basicSalary + hra + da - pf;
    }

    String compareSalary(Employee e) {
        return (this.calculateNetSalary() > e.calculateNetSalary()) ? this.name : e.name;
    }
}

        Employee e2 = new Employee(2, "Rohan", 25000, false);
        Employee e3 = new Employee(3, "Neha", 28000, true);

        System.out.println("Higher between e1 & e2: " + e1.compareSalary(e2));
        System.out.println("Higher between e2 & e3: " + e2.compareSalary(e3));
        System.out.println("Higher between e1 & e3: " + e1.compareSalary(e3));
    }

import java.util.*;

class Student {
    int roll;
    String name;
    int attendance;
    int marks[] = new int[5];

    Student(int roll, String name, int attendance, int[] marks) {
        this.roll = roll;
        this.name = name;
        this.attendance = attendance;
        for (int i = 0; i < 5; i++)
            this.marks[i] = marks[i];
    }

    double getAverage() {
        int sum = 0;
        for (int m : marks) sum += m;
        return sum / 5.0;
    }

    String getResult() {
        double avg = getAverage();
        if (avg >= 60 && attendance >= 75)
            return "Pass";
        else
            return "Fail";
    }

    void show() {
        System.out.println(roll + "  " + name + "  Avg: " + getAverage() + "  Attendance: " + attendance + "  Result: " + getResult());
    }

        this.customerId = customerId;
        this.customerName = customerName;
        this.productName = productName;
        this.quantity = quantity;
        this.pricePerUnit = pricePerUnit;
    }

        System.out.println("Name: " + customerName);
        System.out.println("Product: " + productName);
        System.out.println("Quantity: " + quantity);
        System.out.println("Price/Unit: " + pricePerUnit);
        System.out.println("Discount: " + discountAmount);
        System.out.println("Final Amount: " + finalAmount);
        System.out.println("--------------------------------");
    }
}

class DiscountManager {
    ProductOrder orders[];

    void setOrders(ProductOrder orders[]) {
        this.orders = orders;
    }

    void calculateDiscount() {
        for (ProductOrder o : orders) {
            double total = o.quantity * o.pricePerUnit;

            if (o.quantity > 50)
                o.discountAmount = 0.12 * total;
            else
                o.discountAmount = 0;

            o.finalAmount = total - o.discountAmount;
        }
    }
}

Q5. Write a Java Program to manage 5 library members, where each member can borrow multiple books (count only).
 Rules:
Charging Rules:
First 3 books free
For extra books → ₹25 per book
If member type = “Premium” → 20% discount on total charges
If books > 10 → Extra ₹100 handling charge
Requirements:
Create class Member with:
id, name, memberType (Regular/Premium), totalBooks, charge, discount
Parameterized constructor
Method showDetails()
Create class LibraryManager with:
Method setMembers(Member m[])
Method calculateCharges()


class Member {
    int id;
    String name;
    String memberType;
    int totalBooks;
    double charge;
    double discount;

    Member(int id, String name, String memberType, int totalBooks) {
        this.id = id;
        this.name = name;
        this.memberType = memberType;
        this.totalBooks = totalBooks;
    }

    void showDetails() {
        System.out.println(id + "  " + name + "  Type: " + memberType +
                "  Books: " + totalBooks + "  Charge: " + charge + "  Discount: " + discount);
    }
}

class LibraryManager {
    Member m[];

    void setMembers(Member m[]) {
        this.m = m;
    }

    void calculateCharges() {
        for (Member mem : m) {
            int extraBooks = 0;

            if (mem.totalBooks > 3)
                extraBooks = mem.totalBooks - 3;

            mem.charge = extraBooks * 25;

            if (mem.totalBooks > 10)
                mem.charge += 100;

            if (mem.memberType.equalsIgnoreCase("Premium"))
                mem.discount = 0.20 * mem.charge;
            else
                mem.discount = 0;

            mem.charge -= mem.discount;
        }
    }
}

public class Q5Main {
    public static void main(String[] args) {
        Member arr[] = {
                new Member(1, "Arati", "Premium", 12),
                new Member(2, "Rohan", "Regular", 5),
                new Member(3, "Neha", "Premium", 3)
        };

        LibraryManager lm = new LibraryManager();
        lm.setMembers(arr);
        lm.calculateCharges();

        for (Member m : arr)
            m.showDetails();
    }
}
